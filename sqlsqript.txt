CREATE TABLE role (
    id serial PRIMARY KEY,
    name varchar(50)
);

CREATE TABLE users (
    id serial PRIMARY KEY,
	login varchar(50) NULL,
	password varchar(50) NULL,
	username varchar(50) NULL,
    role_id serial REFERENCES role(id)
);
